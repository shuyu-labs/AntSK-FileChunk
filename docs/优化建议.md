## 1. 增强语义连贯性计算

### 当前问题
- 简单的平均相似度计算
- 未考虑位置权重和语义趋势

### 优化方案
```python
def _enhanced_semantic_coherence(self, current_indices, new_index, embeddings):
    """增强的语义连贯性计算"""
    if not current_indices or new_index >= len(embeddings):
        return 0.0
    
    similarities = []
    weights = []
    new_embedding = embeddings[new_index].reshape(1, -1)
    
    for i, idx in enumerate(current_indices):
        if idx < len(embeddings):
            current_embedding = embeddings[idx].reshape(1, -1)
            similarity = cosine_similarity(new_embedding, current_embedding)[0][0]
            
            # 位置权重：越近的段落权重越高
            position_weight = 0.5 + 0.5 * (i + 1) / len(current_indices)
            
            similarities.append(similarity)
            weights.append(position_weight)
    
    if not similarities:
        return 0.0
    
    # 加权平均
    weighted_coherence = np.average(similarities, weights=weights)
    
    # 语义趋势分析
    if len(similarities) > 2:
        trend_penalty = abs(similarities[-1] - np.mean(similarities[:-1])) * 0.1
        weighted_coherence -= trend_penalty
    
    return max(0.0, min(1.0, weighted_coherence))
```

## 2. 动态阈值调整

### 优化方案
```python
def _adaptive_threshold_strategy(self, embeddings, position):
    """自适应阈值策略"""
    # 基于局部语义密度调整
    local_density = self._calculate_local_semantic_density(embeddings, position)
    
    # 基于文档类型调整
    doc_type_factor = self._detect_document_type_factor(embeddings)
    
    # 基于切片历史调整
    history_factor = self._analyze_chunking_history()
    
    base_threshold = 0.7
    adjusted_threshold = base_threshold * local_density * doc_type_factor * history_factor
    
    return np.clip(adjusted_threshold, 0.4, 0.9)
```

```python
def _detect_semantic_boundaries(self, embeddings, texts):
    """智能语义边界检测"""
    boundaries = []
    
    # 1. 计算语义梯度
    semantic_gradients = []
    for i in range(1, len(embeddings) - 1):
        prev_sim = cosine_similarity(embeddings[i-1:i], embeddings[i:i+1])[0][0]
        next_sim = cosine_similarity(embeddings[i:i+1], embeddings[i+1:i+2])[0][0]
        gradient = abs(prev_sim - next_sim)
        semantic_gradients.append(gradient)
    
    # 2. 寻找梯度峰值
    threshold = np.mean(semantic_gradients) + np.std(semantic_gradients)
    for i, gradient in enumerate(semantic_gradients):
        if gradient > threshold:
            boundaries.append(i + 1)  # 调整索引
    
    # 3. 结构化边界检测
    structural_boundaries = self._detect_structural_boundaries(texts)
    
    # 4. 合并语义和结构边界
    all_boundaries = list(set(boundaries + structural_boundaries))
    return sorted(all_boundaries)
```

## 4. 性能优化

### 向量缓存策略
```python
class EmbeddingCache:
    def __init__(self, max_size=1000):
        from functools import lru_cache
        self.cache = {}
        self.max_size = max_size
    
    def get_embedding(self, text_hash, text, model):
        if text_hash in self.cache:
            return self.cache[text_hash]
        
        embedding = model.encode([text], normalize_embeddings=True)[0]
        
        if len(self.cache) >= self.max_size:
            # 移除最旧的缓存
            oldest_key = next(iter(self.cache))
            del self.cache[oldest_key]
        
        self.cache[text_hash] = embedding
        return embedding
```

### 批量处理优化

```python
def _batch_process_embeddings(self, texts, batch_size=32):
    """批量处理语义向量计算"""
    embeddings = []
    
    for i in range(0, len(texts), batch_size):
        batch = texts[i:i+batch_size]
        batch_embeddings = self.semantic_analyzer.compute_embeddings(batch)
        embeddings.extend(batch_embeddings)
    
    return np.array(embeddings)
```

## 5. 质量评估增强

### 新增评估指标
```python
def _evaluate_boundary_naturalness(self, chunks):
    """评估边界的自然性"""
    boundary_scores = []
    
    for i in range(len(chunks) - 1):
        current_chunk = chunks[i]
        next_chunk = chunks[i + 1]
        
        # 检查边界处的语义连续性
        current_end = current_chunk.content.split('。')[-1]
        next_start = next_chunk.content.split('。')[0]
        
        # 计算边界语义连续性
        boundary_similarity = self.semantic_analyzer.calculate_text_similarity(
            current_end, next_start
        )
        
        # 检查是否在句子中间切断
        sentence_integrity = self._check_sentence_integrity(
            current_chunk.content, next_chunk.content
        )
        
        boundary_score = (1 - boundary_similarity) * sentence_integrity
        boundary_scores.append(boundary_score)
    
    return np.mean(boundary_scores) if boundary_scores else 1.0
```

## 6. 实时调优机制

### 自学习优化
```python
class AdaptiveChunker:
    def __init__(self):
        self.performance_history = []
        self.optimal_configs = {}
    
    def learn_from_feedback(self, chunks, quality_scores):
        """从质量反馈中学习"""
        config_key = self._get_config_key()
        
        if config_key not in self.optimal_configs:
            self.optimal_configs[config_key] = {
                'semantic_threshold': 0.7,
                'target_chunk_size': 800,
                'overlap_ratio': 0.1
            }
        
        # 基于质量分数调整参数
        avg_quality = np.mean(quality_scores)
        
        if avg_quality < 0.7:  # 质量较低，需要调整
            config = self.optimal_configs[config_key]
            
            # 调整策略
            if np.mean([c.semantic_score for c in chunks]) < 0.6:
                config['semantic_threshold'] *= 0.95  # 降低阈值
            
            if np.std([len(c.content) for c in chunks]) > 300:
                config['target_chunk_size'] *= 1.1  # 增加目标大小
        
        self.performance_history.append({
            'config': config_key,
            'quality': avg_quality,
            'timestamp': time.time()
        })
```
